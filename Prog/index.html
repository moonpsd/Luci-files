Estruturas de laço:

While: O while cria uma especie de loop doq estiver dentro da chaves até a condição ser antigida ( ou seja ele vai repetir tudo até condição se tornar VERDADEIRA). 

#include <stdio.h>
int main(void) {
    float A=10, B=2, C=0;

    while(C != A) { // Equanto C for diferente de A repita...
        B = B+1; // Equanto o C for diferente some o valor de B + 1
        printf("B While = %0.1f\n", B);
        C++; // E adicione + 1 no C
    }

    printf("\nB = %0.1f", B);

    return 0;
}

Estruturas de laço:

For: for em C é uma estrutura de repetição ideal quando se sabe exatamente quantas vezes um bloco de código deve ser executado 

#include <stdio.h>
int main(void) {
    int A = 5, B = 2, i;

    // Estrutura: for (INICIALIZAÇÃO; CONDIÇÃO; INCREMENTO)
    for (i = 0; i < A; i++) { 
        B = B + 1;
        printf("I = %d \t B FOR = %d\n", i, B);
    }
    return 0;
}

Expressões numéricas:

#include <stdio.h>
int main()
{
    int A, B, C;

    A = 10;
    B = 20;

    C = A + B; // Soma

    printf("%d\n", C);

    C = A - B; // Subtraçao

    printf("%d\n", C);

    C = A * B; // Multiplicaçao

    printf("%d\n", C);

    C = A / B; // Divisao

    printf("%d\n", C);

    C = A % B; // Resto da divisao entre inteiros

    printf("%d\n", C);

    int X, V;

    X = 10;
    V = 15;

    X++; // Adiciona + 1 ao valor base
    V--; // Subtrai - 1 no valor base

    printf("%d e %d\n", X, V);

    return 0;
}
